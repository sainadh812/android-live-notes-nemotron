cmake_minimum_required(VERSION 3.21)
project(livenotes_desktop_jni LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(JNI REQUIRED)
find_package(Git REQUIRED)
set(TRANSCRIBE_SOURCE_DIR "" CACHE PATH "Pinned transcribe.cpp source checkout")
set(TRANSCRIBE_LIBRARY "" CACHE FILEPATH "Built transcribe import library or shared object")
if(NOT EXISTS "${TRANSCRIBE_SOURCE_DIR}/include/transcribe.h" OR NOT EXISTS "${TRANSCRIBE_LIBRARY}")
    message(FATAL_ERROR "Supply TRANSCRIBE_SOURCE_DIR and TRANSCRIBE_LIBRARY; see build-windows.ps1")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${TRANSCRIBE_SOURCE_DIR}" rev-parse HEAD
    OUTPUT_VARIABLE ENGINE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
if(NOT ENGINE_COMMIT STREQUAL "63a44d9239d610b3908e8a66b384924cd4a77217")
    message(FATAL_ERROR "The JNI ABI requires pinned transcribe.cpp commit 63a44d9239d610b3908e8a66b384924cd4a77217")
endif()
add_library(livenotes_jni SHARED desktop_jni.cpp)
target_include_directories(livenotes_jni PRIVATE ${JNI_INCLUDE_DIRS} include "${TRANSCRIBE_SOURCE_DIR}/include")
target_link_libraries(livenotes_jni PRIVATE "${TRANSCRIBE_LIBRARY}")
if(MSVC)
    target_compile_options(livenotes_jni PRIVATE /utf-8 /EHsc)
endif()
if(UNIX)
    get_filename_component(ENGINE_LIBRARY_DIR "${TRANSCRIBE_LIBRARY}" DIRECTORY)
    set_target_properties(livenotes_jni PROPERTIES BUILD_RPATH "${ENGINE_LIBRARY_DIR};$ORIGIN" INSTALL_RPATH "$ORIGIN")
endif()
install(TARGETS livenotes_jni RUNTIME DESTINATION . LIBRARY DESTINATION .)
